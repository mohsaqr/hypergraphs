
> memberships <- read.csv("../data/memberships.csv")

> diag(together) <- FALSE

+   n_incident_edges = rowSums(crossprod(incidence) > 0) - 1,
+   n_neighbors = colSums(together %*% incidence > 0 & incidence == 0))
